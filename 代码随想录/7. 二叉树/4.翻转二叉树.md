[226. 翻转二叉树 - 力扣（LeetCode）](https://leetcode.cn/problems/invert-binary-tree/description/)

## 前序遍历解法：
```c++
class Solution {
public:
    TreeNode* invertTree(TreeNode* root) {
        if (root == nullptr) {
            return nullptr;
        }
        swap(root->left, root->right); // 交换左右儿子
        invertTree(root->left); // 翻转左子树
        invertTree(root->right); // 翻转右子树
        return root;
    }
};
```
