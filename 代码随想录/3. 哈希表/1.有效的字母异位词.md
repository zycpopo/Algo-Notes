[242. 有效的字母异位词 - 力扣（LeetCode）](https://leetcode.cn/problems/valid-anagram/description/)

## 暴力哈希：
```c++
class Solution {
public:
    bool isAnagram(string s, string t) {
        int hash1[26]={0};
        int hash2[26]={0};
        if(s.size()!=t.size()){
            return false;
        }
        for(auto x :s)
        {
            hash1[x-'a']++;
        }
        for(auto x :t)
        {
            hash2[x-'a']++;
        }
        for(int i=0;i<26;i++){
            if(hash1[i]!=hash2[i]){
                return false;
            }
        }
        return true;
    }
};
```

## 优化版
```c++
#include <string>
using namespace std;
class Solution {
public:

    bool isAnagram(string s, string t) {
        int cnt[26] = {0};
        if (s.size() != t.size()) return false;
        for (char c : s) cnt[c - 'a']++;
        for (char c : t) cnt[c - 'a']--;
        for (int i = 0; i < 26; ++i) {
            if (cnt[i] != 0) return false;
        }
    }
};
```
- **时间复杂度**：O(n)
- **空间复杂度**：O(1)
