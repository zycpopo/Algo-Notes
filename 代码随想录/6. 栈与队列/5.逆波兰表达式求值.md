[150. 逆波兰表达式求值 - 力扣（LeetCode）](https://leetcode.cn/problems/evaluate-reverse-polish-notation/description/)

## 标准题解：
```c++
class Solution {
public:
    int evalRPN(vector<string>& tokens) {
        stack<int> st;
        for (auto& x : tokens) {
            // 判断是运算符（长度为1且是 +-*/）
            // 注意：负数如 "-3" 长度为2，不会误判
            if (x.size() == 1 && (x[0] == '+' || x[0] == '-' || x[0] == '*' || x[0] == '/')) {
                int a = st.top(); st.pop();  // 右操作数（先弹出）
                int b = st.top(); st.pop();  // 左操作数（后弹出）
                
                switch (x[0]) {
                    case '+': st.push(b + a); break;
                    case '-': st.push(b - a); break;
                    case '*': st.push(b * a); break;
                    case '/': st.push(b / a); break;  // 整数除法，向零取整
                }
            } else {
                st.push(stoi(x));  // 数字字符串转 int 压栈
            }
        }
        return st.top();
    }
};
```
这个题很经典了，基础的利用栈解决问题的思想。
