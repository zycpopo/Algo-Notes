```c++
class Solution {
public:
    long long findKthSmallest(vector<int>& coins, int k) {
        int n = coins.size();
        int total = 1 << n;

        vector<long long> lcms(total, 1);

        // 预处理每个子集的 LCM
        for (int mask = 1; mask < total; mask++) {
            for (int i = 0; i < n; i++) {
                if (mask & (1 << i)) {
                    int prev = mask ^ (1 << i);

                    lcms[mask] =
                        lcm(lcms[prev], (long long)coins[i]);

                    break;
                }
            }
        }

        auto count = [&](long long x) {
            long long res = 0;

            for (int mask = 1; mask < total; mask++) {
                if (lcms[mask] > x)
                    continue;

                long long cur = x / lcms[mask];

                // 子集大小是奇数
                if (__builtin_popcount(mask) & 1)
                    res += cur;
                else
                    res -= cur;
            }

            return res;
        };

        long long left = 1;
        long long right =
            1LL * (*min_element(coins.begin(), coins.end())) * k;

        while (left < right) {
            long long mid = left + (right - left) / 2;

            if (count(mid) >= k)
                right = mid;
            else
                left = mid + 1;
        }

        return left;
    }
};
```
