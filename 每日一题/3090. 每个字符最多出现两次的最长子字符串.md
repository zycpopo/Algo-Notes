[3090. 每个字符最多出现两次的最长子字符串 - 力扣（LeetCode）](https://leetcode.cn/problems/maximum-length-substring-with-two-occurrences/?envType=daily-question&envId=2026-08-16)

```c++
class Solution {
public:
    int maximumLengthSubstring(string s) {
        int hash[26] = {0};
        int right = 0;
        int left = 0;
        int result = 0;
        for (right = 0; right < s.size(); right++) {
            hash[s[right] - 'a']++;
            while (hash[s[right] - 'a'] > 2) {
                hash[s[left++] - 'a']--;
            }
            result = max(right - left + 1, result);
        }
        return result;
    }
};
```
