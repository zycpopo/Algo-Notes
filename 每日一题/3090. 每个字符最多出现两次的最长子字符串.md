```c++
class Solution {
public:
    int maximumLengthSubstring(string s) {
        int hash[26] = {0};
        int right = 0;
        int left = 0;
        int result = 0;
        for (right = 0; right < s.size(); right++) {
            hash[s[right] - 'a']++;
            while (hash[s[right] - 'a'] > 2) {
                hash[s[left++] - 'a']--;
            }
            result = max(right - left + 1, result);
        }
        return result;
    }
};
```
