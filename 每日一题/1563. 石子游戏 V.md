class Solution {
public:
    int stoneGameV(vector<int>& stoneValue) {
        int n = stoneValue.size();
        
        // 前缀和，方便快速计算区间和
        vector<int> prefix(n + 1, 0);
        for (int i = 0; i < n; i++) {
            prefix[i + 1] = prefix[i] + stoneValue[i];
        }
        
        // dp[i][j] 表示区间 [i, j] 能获得的最大分数
        vector<vector<int>> dp(n, vector<int>(n, 0));
        
        // 区间长度从 2 到 n
        for (int len = 2; len <= n; len++) {
            for (int i = 0; i + len - 1 < n; i++) {
                int j = i + len - 1;
                
                // 枚举分割点 k
                for (int k = i; k < j; k++) {
                    int sumLeft = prefix[k + 1] - prefix[i];
                    int sumRight = prefix[j + 1] - prefix[k + 1];
                    
                    int currentScore;
                    if (sumLeft < sumRight) {
                        // 左边小，留下左边
                        currentScore = sumLeft + dp[i][k];
                    } else if (sumLeft > sumRight) {
                        // 右边小，留下右边
                        currentScore = sumRight + dp[k + 1][j];
                    } else {
                        // 相等，Alice 可以选择留下哪边，取较大的后续收益
                        currentScore = sumLeft + max(dp[i][k], dp[k + 1][j]);
                    }
                    
                    dp[i][j] = max(dp[i][j], currentScore);
                }
            }
        }
        
        return dp[0][n - 1];
    }
};
