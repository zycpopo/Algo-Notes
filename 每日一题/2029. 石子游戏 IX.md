[2029. 石子游戏 IX - 力扣（LeetCode）](https://leetcode.cn/problems/stone-game-ix/description/?envType=daily-question&envId=2026-08-16)

```cpp
class Solution {
public:
    bool stoneGameIX(vector<int>& stones) {
        int cnt[3]{};
        for (int x : stones) {
            cnt[x % 3]++;        }
        if (cnt[0] % 2 == 0) {
            return cnt[1] > 0 && cnt[2] > 0;
        }
        return abs(cnt[1] - cnt[2]) > 2;
    }
};
```
