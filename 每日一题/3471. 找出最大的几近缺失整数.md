class Solution {
public:
    int f(vector<int>& nums, int begin, int end, int x) {
        if (find(nums.begin() + begin, nums.begin() + end, x) !=
            nums.begin() + end) {
            return -1;
        } else
            return x;
    }
    int largestInteger(vector<int>& nums, int k) {
        int n = nums.size();
        if (k == n) {
            return ranges::max(nums);
        }
        if (k == 1) {
            unordered_map<int, int> cnt;
            for (int x : nums) {
                cnt[x]++;
            }
            int res = -1;
            for (auto& [x, c] : cnt) {
                if (c == 1) {
                    res = max(res, x);
                }
            }
            return res;
        } else {
            return max(f(nums, 1, n, nums[0]), f(nums, 0, n - 1, nums[n - 1]));
        }
    }
};
