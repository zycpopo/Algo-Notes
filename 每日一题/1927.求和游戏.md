class Solution {
public:
    bool sumGame(string num) {
        int n = num.size();
        int half = n / 2;
        
        int diff = 0;  // 前一半和 - 后一半和
        int leftQ = 0, rightQ = 0;
        
        // 计算差值
        for (int i = 0; i < n; i++) {
            if (num[i] == '?') {
                if (i < half) leftQ++;
                else rightQ++;
            } else {
                if (i < half) diff += num[i] - '0';
                else diff -= num[i] - '0';
            }
        }
        
        // 关键判断
        return diff * 2 + 9 * (leftQ - rightQ) != 0;
    }
};
